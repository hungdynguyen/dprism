Work orchestration05 / EXPERT KNOWLEDGE REPLICATION

Một component mỗi lượt; nhiều expert song song

Chọn task PENDING có mọi dependency DONE. Fan-out chỉ diễn ra giữa các primary của component đã chọn.

PENDING → ASSESSING

Task đủ điều kiện; đặt trạng thái trước khi phát Send.

REVIEWING

Barrier đủ bài; review theo mode và giới hạn round.

DONE / SKIPPED

DONE có brief. SKIPPED có lý do và lan truyền tới task phụ thuộc.

WAIT / BLOCKED

Đang chạy: chờ. Không thể tiến tiếp: interrupt, không quay vòng bận.

STATE & ISOLATION

Hai cấp song song

Phiên bản hiện tại: sequential_components. Nâng cấp song song giữa component cần state và reducer theo component ID.

Send mang payload riêng; worker phải tôn trọng namespace memory và quyền truy xuất. Không cấp quyền đọc draft của primary khác.

Execution contracts

State khởi tạo: mọi task PENDING; briefs, assessments, skip_reasons và component_tokens rỗng; spent_tokens=0. Expert catalog được nạp theo phiên bản.

Worker trả assessment theo key component:expert:round. Barrier đối chiếu tập expert ID, không chỉ đếm số bản ghi.

Token ledger cấp phát nguyên tử trước mỗi LLM call, bao gồm retry, consulted và delegates.

PYTHON · SCHEDULER KERNEL + SEND + REDUCER
from copy import deepcopy
from typing import Annotated, TypedDict
from langgraph.types import Send, interrupt

TERMINAL = {"DONE", "SKIPPED"}
IN_FLIGHT = {"ASSESSING", "REVIEWING", "SYNTHESIZING", "WAITING_HUMAN"}

def merge_assessments(left, right):
    merged = dict(left)
    for key, value in right.items():
        # key = component_id:expert_id:round; identical retries are idempotent.
        if key in merged and merged[key] != value:
            raise ValueError("Conflicting output for the same assessment key")
        merged[key] = value
    return merged

class State(TypedDict):
    plan: dict
    experts: dict
    status: dict
    briefs: dict
    spent_tokens: int
    component_tokens: dict
    case_id: str
    dispatch: list
    route: str
    skip_reasons: dict
    assessments: Annotated[dict, merge_assessments]

def resolve_component(plan, component):
    resolved = deepcopy(plan["component_defaults"])
    for key, value in component.items():
        if key == "policy_overrides":
            resolved["policy"].update(deepcopy(value))
        elif key == "assignments":
            resolved["assignments"].update(deepcopy(value))
        else:
            resolved[key] = deepcopy(value)
    return resolved

def evaluate_when(condition, briefs):
    if condition is None:
        return True
    if condition["field"] != "rating" or condition["operator"] != "IN":
        raise ValueError("Unsupported predicate")
    return briefs[condition["component"]]["rating"] in condition["value"]

def schedule(state):
    plan = state["plan"]  # immutable artifact accepted by validate_plan
    status = dict(state["status"])
    reasons = dict(state["skip_reasons"])
    components = [resolve_component(plan, c) for c in plan["components"]]
    # Propagate SKIPPED even when catalog order is not topological.
    changed = True
    while changed:
        changed = False
        for c in components:
            if status[c["id"]] != "PENDING":
                continue
            deps = c["depends_on"]
            if not all(status[d] in TERMINAL for d in deps):
                continue
            skipped_parent = any(status[d] == "SKIPPED" for d in deps)
            if skipped_parent or not evaluate_when(c.get("when"), state["briefs"]):
                status[c["id"]] = "SKIPPED"
                reasons[c["id"]] = "dependency_skipped" if skipped_parent else "when_false"
                changed = True

    base = {"status": status, "skip_reasons": reasons, "dispatch": []}
    if all(v in TERMINAL for v in status.values()):
        return {**base, "route": "assemble"}
    if any(v in IN_FLIGHT for v in status.values()):
        return {**base, "route": "wait"}  # sequential_components
    ready = [c for c in components
             if status[c["id"]] == "PENDING"
             and all(status[d] == "DONE" for d in c["depends_on"])]
    if not ready:
        return {**base, "route": "blocked"}
    c = ready[0]  # deterministic catalog order; one component at a time
    global_left = plan["global_execution_policies"]["total_token_budget"] - state["spent_tokens"]
    component_left = c["policy"]["token_budget"] - state["component_tokens"].get(c["id"], 0)
    if min(global_left, component_left) <= 0:
        return {**base, "route": "budget"}
    primaries = c["assignments"]["primaries"]
    if not primaries:
        raise ValueError("Every component needs at least one primary")
    upstream = {d: deepcopy(state["briefs"][d]) for d in c["depends_on"]}
    payloads = []
    for expert_id in primaries:
        expert = state["experts"][expert_id]  # trusted catalog snapshot
        payloads.append({
            "case_id": state["case_id"], "component_id": c["id"],
            "question": c["name"], "expert_id": expert_id, "round": 0,
            "kb_version": expert["kb_version"],
            "source_allowlist": deepcopy(expert["source_allowlist"]),
            "scope_guard": deepcopy(c["scope_guard"]),
            "policy": deepcopy(c["policy"]),
            "coordination_pattern": c["coordination_pattern"],
            "orchestration_spec": deepcopy(c.get("orchestration_spec")),
            "consulted": list(c["assignments"]["consulted"]),
            "memory_namespace": f'{state["case_id"]}:{c["id"]}:{expert_id}:0',
            "upstream": deepcopy(upstream),
        })
    status[c["id"]] = "ASSESSING"
    return {**base, "status": status, "route": "assess", "dispatch": payloads}

def route_schedule(state):
    if state["route"] == "assemble":
        return "assemble"
    if state["route"] == "assess":
        return [Send("assess_one", payload) for payload in state["dispatch"]]
    # Durable pause, not a polling loop. Resume only after state is reconciled.
    interrupt({"reason": state["route"], "status": state["status"]})
    return "schedule"

def wire_scheduler(builder):
    builder.add_node("schedule", schedule)
    builder.add_conditional_edges("schedule", route_schedule)
    # assess_one and assemble are registered by the enclosing fixed graph.
    return builder

Decision engine07 / EXPERT KNOWLEDGE REPLICATION

Tổng hợp có cấu trúc; gate kiểm tra trước phát hành

REVIEW ≠ RELEASE

Một vị trí cho mỗi quyết định

after_assess: chọn review theo mode và số primary.

after_review: Δ≤1 là hội tụ theo policy; Δ>1 chỉ lặp khi còn lượt. Hết lượt vẫn giữ bất đồng.

gate: sau synthesis, kiểm tra schema, scope, citation, Δ cuối, confidence và CRITICAL.

Rating và bảo tồn bất đồng

LOW=0 · MOD=1 · HIGH=2 · CRITICAL=3. Với N bài mới nhất: Δ=max−min; với hai bài: abs(rA−rB).

Chọn mức cao hơn khi có bằng chứng; thiếu bằng chứng ghi uncertainty và chuyển human. Dissent chỉ lưu một lần; brief tham chiếu ID.

synthesize tạo brief; commit lưu brief đã duyệt cùng phiên bản. Human resolution lưu trace trước sửa và quyết định override ngưỡng; không bỏ qua lỗi schema/scope/citation.

PYDANTIC 2 · OUTPUT CONTRACT + DELTA / RELEASE RULES
from typing import Literal
from pydantic import BaseModel, ConfigDict, Field, model_validator

Rating = Literal["LOW", "MOD", "HIGH", "CRITICAL"]
SCORE = {"LOW": 0, "MOD": 1, "HIGH": 2, "CRITICAL": 3}

class Contract(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)

class Provenance(Contract):
    expert_id: str = Field(min_length=1)
    kb_version: str = Field(min_length=1)
    prompt_version: str = Field(min_length=1)
    model_snapshot: str = Field(min_length=1)

class Evidence(Contract):
    evidence_id: str = Field(min_length=1)
    source_id: str = Field(min_length=1)
    source_version: str = Field(min_length=1)
    locator: str = Field(min_length=1)
    excerpt: str = Field(min_length=1)
    sha256: str = Field(pattern=r"^[a-f0-9]{64}$")

class Finding(Contract):
    finding_id: str = Field(min_length=1)
    claim: str = Field(min_length=1)
    scope_tag: str = Field(min_length=1)
    evidence_ids: list[str] = Field(min_length=1)
    author_expert_ids: list[str] = Field(min_length=1)

class FindingSet(Contract):
    key_findings: list[Finding] = Field(min_length=1)
    evidence_cited: list[Evidence] = Field(min_length=1)

    @model_validator(mode="after")
    def check_references(self):
        ids = [e.evidence_id for e in self.evidence_cited]
        finding_ids = [f.finding_id for f in self.key_findings]
        if len(ids) != len(set(ids)) or len(finding_ids) != len(set(finding_ids)):
            raise ValueError("Duplicate evidence or finding IDs")
        if any(not set(f.evidence_ids) <= set(ids) for f in self.key_findings):
            raise ValueError("Finding references unknown evidence")
        return self

class Assessment(FindingSet):
    component_id: str = Field(min_length=1)
    round: int = Field(ge=0)
    provenance: Provenance
    rating: Rating
    rationale: str = Field(min_length=1)
    confidence: float = Field(ge=0, le=1)

class Disagreement(Contract):
    disagreement_id: str = Field(min_length=1)
    aspect: str = Field(min_length=1)
    expert_a_id: str = Field(min_length=1)
    expert_a_view: str = Field(min_length=1)
    expert_b_id: str = Field(min_length=1)
    expert_b_view: str = Field(min_length=1)
    resolution_note: str = Field(min_length=1)
    status: Literal["RESOLVED", "UNRESOLVED"]

class UpstreamBrief(Contract):
    component_id: str = Field(min_length=1)
    brief_version: int = Field(ge=1)
    rating: Rating
    conclusion: str = Field(min_length=1, max_length=1000)
    finding_ids: list[str]
    evidence_ids: list[str]
    uncertainties: list[str]
    dissent_ids: list[str]
    provenance: list[Provenance] = Field(min_length=1)

class Synthesis(FindingSet):
    component_id: str = Field(min_length=1)
    rating: Rating
    confidence: float = Field(ge=0, le=1)
    rationale: str = Field(min_length=1)
    disagreements: list[Disagreement]
    # IDs of unresolved disagreements; views are stored once, above.
    dissenting_views: list[str]
    upstream_brief: UpstreamBrief

    @model_validator(mode="after")
    def check_brief_and_dissent(self):
        b = self.upstream_brief
        if (b.component_id, b.rating) != (self.component_id, self.rating):
            raise ValueError("Brief and synthesis disagree")
        ids = [d.disagreement_id for d in self.disagreements]
        if len(ids) != len(set(ids)):
            raise ValueError("Duplicate disagreement IDs")
        unresolved = {d.disagreement_id for d in self.disagreements
                      if d.status == "UNRESOLVED"}
        if set(self.dissenting_views) != unresolved or set(b.dissent_ids) != unresolved:
            raise ValueError("Unresolved dissent must be preserved")
        if set(b.finding_ids) != {f.finding_id for f in self.key_findings}:
            raise ValueError("Brief finding index is incomplete")
        if set(b.evidence_ids) != {e.evidence_id for e in self.evidence_cited}:
            raise ValueError("Brief evidence index is incomplete")
        authors = {p.expert_id for p in b.provenance}
        if any(not set(f.author_expert_ids) <= authors for f in self.key_findings):
            raise ValueError("Missing finding provenance")
        if any(d.expert_a_id not in authors or d.expert_b_id not in authors
               for d in self.disagreements):
            raise ValueError("Missing dissent provenance")
        return self

def rating_delta(latest_assessments):
    values = [SCORE[a.rating] for a in latest_assessments]
    if not values:
        raise ValueError("No assessments: barrier must block")
    return max(values) - min(values)  # one primary: 0; two: abs(rA-rB)

def release_reasons(synthesis, latest_assessments, rules):
    reasons = []
    if rating_delta(latest_assessments) >= rules["rating_delta_threshold"]:
        reasons.append("rating_delta")
    if synthesis.confidence < rules["confidence_minimum"]:
        reasons.append("low_confidence")
    if rules["escalate_on_critical"] and synthesis.rating == "CRITICAL":
        reasons.append("critical")
    return reasons
